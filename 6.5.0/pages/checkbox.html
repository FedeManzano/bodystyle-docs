✅ Checkbox

Los checkboxes o casillas de verificación son elementos de formulario que permiten a los usuarios seleccionar una o más opciones de un conjunto. A diferencia de los botones de opción (radio buttons), los checkboxes permiten múltiples selecciones simultáneas.

Muestra

Vamos a ver una muestra de como utilizar los checkboxes en Bodystyle.

Colores

Bodystyle ofrece una variedad de colores predefinidos para personalizar la apariencia de los checkboxes. Estos colores permiten adaptar los elementos a diferentes temas y estilos de diseño, mejorando la experiencia visual del usuario.

     // Validación del formulario
    const formulario = document.getElementById('formularioRegistro');
    const camposValidacion = ['dni', 'nombre', 'apellido', 'email', 'telefono', 'password', 'repetir'];
    function validarEmail(email) {
        const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        return regex.test(email);
    }

    function marcarError(inputId) {
        const input = document.getElementById(inputId);
        input.classList.add('i-error');
    }

    function limpiarError(inputId) {
        const input = document.getElementById(inputId);
        input.classList.remove('i-error');
    }
    
    function validarFormulario() {
        let valido = true;
        // Validar campos vacíos
        camposValidacion.forEach(id => {
            const input = document.getElementById(id);
            if (!input.value.trim()) {
                marcarError(id);
                valido = false;
            } else {
                limpiarError(id);
            }
        });
        // Validar email
        const email = document.getElementById('email');
        if (email.value.trim() && !validarEmail(email.value)) {
            marcarError('email');
            valido = false;
        } else {
            limpiarError('email');
        }
        // Validar que las contraseñas coincidan
        const password = document.getElementById('password');
        const repetir = document.getElementById('repetir');
        if (password.value !== repetir.value) {
            marcarError('repetir');
            valido = false;
        } else {
            limpiarError('repetir');
        }
        // Validar checkbox de contrato
        const contrato = document.getElementById('contrato');
        if (!contrato.checked) {
            valido = false;
        }
        return valido;
    }
    // Event listener para el envío del formulario
    formulario.addEventListener('submit', function(e) {
        e.preventDefault();
        if (validarFormulario()) {
            // Mostrar toast de éxito
            BS.Toast({
                html: "✅ Formulario enviado correctamente",
                clases: ["bg-green", "bor-rad-10", "c-white"],
                tiempo: 3000,
                cerrar: true
            });
            // Opcional: limpiar formulario
            setTimeout(() => {
                formulario.reset();
                camposValidacion.forEach(id => limpiarError(id));
            }, 500);
        } else {
            // Mostrar toast de error
            BS.Toast({
                html: "❌ Por favor, completa todos los campos correctamente",
                clases: ["bg-red", "bor-rad-10", "c-white"],
                tiempo: 3000,
                cerrar: true
            });
        }
    });
    // Limpiar error al escribir
    camposValidacion.forEach(id => {
        const input = document.getElementById(id);
        if (input) {
            input.addEventListener('input', function() {
                limpiarError(id);
            });
        }
    });
